 16"""
 17Beta Metadata Patcher Script
 18
 19Generates standardized SQL patch files for fixing metadata issues in:
 201. Production metadata database (ensembl_genome_metadata)
 212. Core database (meta table)
 22
 23This ensures metadata consistency between both databases and provides
 24a standardized workflow for creating and applying patches.
 25
 26Requires genome_uuid for all patches.
 27
 28Usage:
 29    python beta_patcher.py patches.csv --jira-ticket EBD-1111 --output-dir ./patches/
 30    python beta_patcher.py patches.csv --jira-ticket EBD-1111 --core-suffix _core_115_1
 31"""

 73def get_genome_adaptor() -> Optional[GenomeAdaptor]:
 74    """
 75    Create a GenomeAdaptor instance using environment variables.
 76
 77    Requires:
 78        METADATA_URI: Connection string for metadata database
 79        TAXONOMY_URI: Connection string for taxonomy database
 80
 81    Returns:
 82        GenomeAdaptor instance or None if not available
 83    """
 84    if not METADATA_API_AVAILABLE:
 85        logging.error("ensembl-metadata-api package not available")
 86        return None
 87
 88    metadata_uri = os.getenv("METADATA_URI")
 89    taxonomy_uri = os.getenv("TAXONOMY_URI")
 90
 91    if not metadata_uri or not taxonomy_uri:
 92        logging.error(
 93            "METADATA_URI and TAXONOMY_URI environment variables required. "
 94            "Format: mysql+pymysql://user:pass@host:port/database"
 95        )
 96        return None
 97
 98    try:
 99        return GenomeAdaptor(metadata_uri=metadata_uri, taxonomy_uri=taxonomy_uri)
100    except Exception as e:
101        logging.error(f"Failed to create GenomeAdaptor: {e}")
102        return None

105def get_genome_by_uuid(genome_uuid: str) -> Optional[Dict]:
106    """
107    Fetch genome details by UUID using ensembl-metadata-api.
108
109    Args:
110        genome_uuid: Genome UUID to query
111
112    Returns:
113        Dict containing genome information or None if not found
114    """
115    adaptor = get_genome_adaptor()
116    if not adaptor:
117        return None
118
119    try:
120        results = adaptor.fetch_genomes_by_genome_uuid(genome_uuid=genome_uuid)
121        if results:
122            genome, organism, assembly, release, site = results[0]
123            return {
124                "genome_uuid": genome.genome_uuid,
125                "production_name": genome.production_name,
126                "genebuild_version": genome.genebuild_version,
127                "genebuild_date": genome.genebuild_date,
128                "organism": {
129                    "organism_uuid": organism.organism_uuid,
130                    "taxonomy_id": organism.taxonomy_id,
131                    "scientific_name": organism.scientific_name,
132                    "strain": organism.strain,
133                    "biosample_id": organism.biosample_id,
134                },
135                "assembly": {
136                    "assembly_uuid": assembly.assembly_uuid,
137                    "accession": assembly.accession,
138                    "name": assembly.name,
139                    "level": assembly.level,
140                },
141            }
142        return None
143    except Exception as e:
144        logging.error(f"Failed to fetch genome {genome_uuid}: {e}")
145        return None

148def get_team_responsible_for_genome(
149    genome_uuid: str, dataset_type: str = "genebuild"
150) -> Optional[str]:
151    """
152    Fetch genebuild.team_responsible attribute for a genome.
153
154    Args:
155        genome_uuid: Genome UUID to query
156        dataset_type: Dataset type (default: genebuild)
157
158    Returns:
159        Team responsible string or None if not found
160    """
161    adaptor = get_genome_adaptor()
162    if not adaptor:
163        return None
164
165    try:
166        query = text(
167            """
168            SELECT dataset_attribute.value
169            FROM dataset_attribute
170            JOIN attribute ON dataset_attribute.attribute_id = attribute.attribute_id
171            JOIN dataset ON dataset_attribute.dataset_id = dataset.dataset_id
172            JOIN genome_dataset ON dataset.dataset_id = genome_dataset.dataset_id
173            JOIN genome ON genome_dataset.genome_id = genome.genome_id
174            WHERE genome.genome_uuid = :genome_uuid
175            AND dataset.name = :dataset_type
176            AND attribute.name = 'genebuild.team_responsible'
177        """
178        )
179
180        with adaptor.metadata_db.connect() as conn:
181            result = conn.execute(
182                query, {"genome_uuid": genome_uuid, "dataset_type": dataset_type}
183            )
184            row = result.fetchone()
185            if row:
186                return row[0]
187        return None
188    except Exception as e:
189        logging.warning(
190            f"Failed to fetch team_responsible for genome {genome_uuid}: {e}"
191        )
192        return None

195def get_affected_genomes_and_teams(
196    genome_uuid: str, table_location: str, dataset_type: str = "genebuild"
197) -> List[Dict]:
198    """
199    Get all genomes that share the same organism or assembly, and their team_responsible values.
200
201    Args:
202        genome_uuid: Source genome UUID
203        table_location: Table being modified (organism or assembly)
204        dataset_type: Dataset type (default: genebuild)
205
206    Returns:
207        List of dicts with genome_uuid, production_name, and team_responsible
208    """
209    if table_location not in ["organism", "assembly"]:
210        return []
211
212    adaptor = get_genome_adaptor()
213    if not adaptor:
214        return []
215
216    try:
217        if table_location == "organism":
218            # Find all genomes sharing the same organism
219            query = text(
220                """
221                SELECT g2.genome_uuid, g2.production_name
222                FROM genome g1
223                JOIN genome g2 ON g1.organism_id = g2.organism_id
224                WHERE g1.genome_uuid = :genome_uuid
225            """
226            )
227        else:  # assembly
228            # Find all genomes sharing the same assembly
229            query = text(
230                """
231                SELECT g2.genome_uuid, g2.production_name
232                FROM genome g1
233                JOIN genome g2 ON g1.assembly_id = g2.assembly_id
234                WHERE g1.genome_uuid = :genome_uuid
235            """
236            )
237
238        affected_genomes = []
239        with adaptor.metadata_db.connect() as conn:
240            result = conn.execute(query, {"genome_uuid": genome_uuid})
241            for row in result:
242                uuid, prod_name = row
243                team = get_team_responsible_for_genome(uuid, dataset_type)
244                affected_genomes.append(
245                    {
246                        "genome_uuid": uuid,
247                        "production_name": prod_name,
248                        "team_responsible": team or "UNKNOWN",
249                    }
250                )
251
252        return affected_genomes
253    except Exception as e:
254        logging.warning(f"Failed to fetch affected genomes for {genome_uuid}: {e}")
255        return []

259def _metadata_db_joins() -> str:
260    """Generate standardized JOIN clauses for metadata DB queries."""
261    return (
262        "JOIN attribute ON dataset_attribute.attribute_id = attribute.attribute_id\n"
263        "JOIN dataset ON dataset_attribute.dataset_id = dataset.dataset_id\n"
264        "JOIN genome_dataset ON dataset.dataset_id = genome_dataset.dataset_id\n"
265        "JOIN genome ON genome_dataset.genome_id = genome.genome_id"
266    )

269def _metadata_db_where(genome_uuid: str, dataset_type: str, attribute_name: str) -> str:
270    """Generate standardized WHERE clause for metadata DB queries."""
271    return (
272        f"WHERE genome.genome_uuid = '{genome_uuid}'\n"
273        f"AND dataset.name = '{dataset_type}'\n"
274        f"AND attribute.name = '{attribute_name}'"
275    )

278def _log_team_warnings(  # pylint: disable=too-many-locals, too-many-arguments
279    validate_file,
280    patch_file,
281    genome_uuid: str,
282    affected_genomes: List[Dict],
283    team_filter: Optional[str],
284    logger: Optional[logging.Logger],
285):
286    """
287    Logs warnings about affected genomes and team ownership.
288    """
289    teams_map: Dict[str, List[str]] = {}
290    for genome_data in affected_genomes:
291        team = genome_data["team_responsible"]
292        teams_map.setdefault(team, []).append(
293            f"{genome_data['production_name']} ({genome_data['genome_uuid']})"
294        )
295
296    teams_set = set(teams_map.keys())
297    skip_due_to_filter = False
298    if team_filter and teams_set != {team_filter}:
299        skip_due_to_filter = True
300        msg = f"""
301-- SKIPPED: Modification affects teams {teams_set}, filter requires only '{team_filter}'
302"""
303        validate_file.write(msg)
304        patch_file.write(msg)
305        for team, genomes in teams_map.items():
306            validate_file.write(f"--   Team: {team}\n")
307            patch_file.write(f"--   Team: {team}\n")
308            for genome_str in genomes:
309                line = f"--     - {genome_str}\n"
310                validate_file.write(line)
311                patch_file.write(line)
312        validate_file.write("\n")
313        patch_file.write("\n")
314        if logger:
315            logger.warning(
316                f"""
317                SKIPPED: {genome_uuid} - affects teams {teams_set},
318                filter requires '{team_filter}'
319                """
320            )
321    else:
322        # Normal warning header
323        header = f"""
324        -- WARNING: Modifying shared entity affects {len(affected_genomes)} genome(s)\n
325        """
326        validate_file.write(header)
327        patch_file.write(header)
328        for team, genomes in teams_map.items():
329            validate_file.write(f"--   Team: {team}\n")
330            patch_file.write(f"--   Team: {team}\n")
331            for genome_str in genomes:
332                line = f"--     - {genome_str}\n"
333                validate_file.write(line)
334                patch_file.write(line)
335
336        if team_filter:
337            note = f"-- Team filter '{team_filter}' PASSED - all affected owned by this team\n"
338            validate_file.write(note)
339            patch_file.write(note)
340        else:
341            note = "-- ENSURE CONSULTATION WITH ALL AFFECTED TEAMS BEFORE APPLYING\n"
342            validate_file.write(note)
343            patch_file.write(note)
344
345    return skip_due_to_filter

348def _write_validation_sql(  # pylint: disable= too-many-arguments
349    validate_file,
350    genome_uuid: str,
351    attribute_name: str,
352    escaped_value: str,
353    table_location: str,
354    dataset_type: str,
355):
356    """Writes the validation SELECT statements for a patch."""
357    validate_file.write(
358        f"-- {genome_uuid} | {attribute_name} (table: {table_location})\n"
359    )
360    if table_location == "dataset_attribute":
361        validate_file.write(
362            "SELECT genome.genome_uuid, dataset_attribute.value AS current_value, "
363            f"'{escaped_value}' AS proposed_value\nFROM dataset_attribute\n"
364        )
365        validate_file.write(_metadata_db_joins() + "\n")
366        validate_file.write(
367            _metadata_db_where(genome_uuid, dataset_type, attribute_name) + ";\n\n"
368        )
369    else:
370        column_name = attribute_name.split(".")[-1]
371        table_map = {
372            "genome": "genome",
373            "organism": "organism JOIN genome ON organism.organism_id = genome.organism_id",
374            "assembly": "assembly JOIN genome ON assembly.assembly_id = genome.assembly_id",
375        }
376        validate_file.write(
377            f"""
378            SELECT 
379                genome.genome_uuid, {table_location}.{column_name} AS current_value,
380                '{escaped_value}' AS proposed_value\n
381            """
382        )
383        validate_file.write(f"FROM {table_map[table_location]}\n")
384        validate_file.write(f"WHERE genome.genome_uuid = '{genome_uuid}';\n\n")

387def _write_patch_sql(  # pylint: disable=too-many-arguments
388    patch_file,
389    genome_uuid: str,
390    attribute_name: str,
391    escaped_value: str,
392    table_location: str,
393    dataset_type: str,
394):
395    """Writes the patch UPDATE/INSERT statements for a patch."""
396    patch_file.write(f"-- {genome_uuid} | {attribute_name} (table: {table_location})\n")
397    if table_location == "dataset_attribute":
398        # DELETE existing attribute value
399        patch_file.write("DELETE dataset_attribute FROM dataset_attribute\n")
400        patch_file.write(_metadata_db_joins() + "\n")
401        patch_file.write(
402            _metadata_db_where(genome_uuid, dataset_type, attribute_name) + ";\n\n"
403        )
404
405        # INSERT new value
406        patch_file.write(
407            "INSERT INTO dataset_attribute (dataset_id, attribute_id, value)\n"
408        )
409        patch_file.write(
410            "SELECT dataset.dataset_id, attribute.attribute_id, " f"'{escaped_value}'\n"
411        )
412        patch_file.write("FROM dataset\n")
413        patch_file.write(
414            "JOIN genome_dataset ON dataset.dataset_id = genome_dataset.dataset_id\n"
415        )
416        patch_file.write("JOIN genome ON genome_dataset.genome_id = genome.genome_id\n")
417        patch_file.write("JOIN attribute ON attribute.name = " f"'{attribute_name}'\n")
418        patch_file.write(
419            f"WHERE genome.genome_uuid = '{genome_uuid}'\nAND dataset.name = '{dataset_type}';\n\n"
420        )
421    else:
422        column_name = attribute_name.split(".")[-1]
423        table_map = {
424            "genome": "UPDATE genome SET",
425            "organism": """
426            UPDATE organism JOIN genome
427            ON organism.organism_id = genome.organism_id SET
428            """,
429            "assembly": """
430            UPDATE assembly JOIN genome 
431            ON assembly.assembly_id = genome.assembly_id SET
432            """,
433        }
434        patch_file.write(
435            f"{table_map[table_location]} {table_location}.{column_name} = '{escaped_value}'\n"
436        )
437        patch_file.write(f"WHERE genome.genome_uuid = '{genome_uuid}';\n\n")

440def write_metadata_patch_for_genome(  # pylint: disable=too-many-arguments
441    validate_file,
442    patch_file,
443    genome_uuid: str,
444    patches: List[Tuple[str, str, str]],
445    dataset_type: str = "genebuild",
446    logger: Optional[logging.Logger] = None,
447    team_filter: Optional[str] = "Genebuild",
448):
449    """
450    Write validation and patch SQL for a single genome.
451    """
452    valid_tables = ["dataset_attribute", "genome", "organism", "assembly"]
453    for attribute_name, new_value, table_location in patches:
454        escaped_value = new_value.replace("'", "''")
455
456        if table_location not in valid_tables:
457            msg = f"-- ERROR: Invalid table_location '{table_location}' for {attribute_name}\n"
458            validate_file.write(msg)
459            patch_file.write(msg)
460            continue
461
462        # Organism/assembly checks
463        if table_location in ["organism", "assembly"]:
464            affected_genomes = get_affected_genomes_and_teams(
465                genome_uuid, table_location, dataset_type
466            )
467            if affected_genomes:
468                skip = _log_team_warnings(
469                    validate_file,
470                    patch_file,
471                    genome_uuid,
472                    affected_genomes,
473                    team_filter,
474                    logger,
475                )
476                if skip:
477                    continue
478
479        # Write SQL
480        _write_validation_sql(
481            validate_file,
482            genome_uuid,
483            attribute_name,
484            escaped_value,
485            table_location,
486            dataset_type,
487        )
488        _write_patch_sql(
489            patch_file,
490            genome_uuid,
491            attribute_name,
492            escaped_value,
493            table_location,
494            dataset_type,
495        )

498def _core_db_where(meta_key: str, species_id: int) -> str:
499    """Generate standardized WHERE clause for core DB queries."""
500    return f"WHERE meta_key = '{meta_key}'\n" f"AND species_id = {species_id}"

503def write_core_patch_for_genome(
504    validate_file,
505    patch_file,
506    database: str,
507    patches: List[Tuple[str, str, str]],
508    species_id: int = 1,
509):
510    """
511    Write validation and patch SQL for a single core database to open file handles.
512
513    Args:
514        validate_file: Open file handle for validation SQL
515        patch_file: Open file handle for patch SQL
516        database: Core database name
517        patches: List of (meta_key, new_value, table_location) \
518            tuples (table_location ignored for core DB)
519        species_id: Species ID
520    """
521    for meta_key, new_value, _table_location in patches:
522        escaped_value = new_value.replace("'", "''")
523
524        # Write validation SQL
525        validate_file.write(f"-- {database} | {meta_key}\n")
526        validate_file.write(f"USE {database};\n")
527        validate_file.write(f"SELECT '{database}' AS database_name, ")
528        validate_file.write(f"'{meta_key}' AS meta_key, ")
529        validate_file.write("meta_value AS current_value, ")
530        validate_file.write(f"'{escaped_value}' AS proposed_value\n")
531        validate_file.write("FROM meta\n")
532        validate_file.write(_core_db_where(meta_key, species_id))
533        validate_file.write(";\n\n")
534
535        # Write patch SQL
536        patch_file.write(f"-- {database} | {meta_key}\n")
537        patch_file.write(f"USE {database};\n")
538        patch_file.write("UPDATE meta\n")
539        patch_file.write(f"SET meta_value = '{escaped_value}'\n")
540        patch_file.write(_core_db_where(meta_key, species_id))
541        patch_file.write(";\n\n")
542        patch_file.write("INSERT IGNORE INTO meta (species_id, meta_key, meta_value)\n")
543        patch_file.write(f"VALUES ({species_id}, '{meta_key}', '{escaped_value}');\n\n")

546def read_csv_patches(csv_file: Path, core_suffix: str = "_core_114_1") -> List[Dict]:
547    """
548    Read patches from CSV file.
549
550    CSV columns:
551    - genome_uuid (required)
552    - meta_key (required)
553    - desired_meta_value (required)
554    - dataset_type (optional, default: genebuild)
555    - species_id (optional, default: 1)
556    - table_location (optional, default: dataset_attribute)
557      Valid values: dataset_attribute, genome, organism, assembly
558
559    Args:
560        csv_file: Path to CSV file
561        core_suffix: Suffix to append to production_name for core DB name
562
563    Returns:
564        List of patch dictionaries
565
566    Raises:
567        ValueError if CSV format is invalid
568    """
569    patches = []
570
571    with open(csv_file, "r") as f:
572        reader = csv.DictReader(f)
573
574        # Validate required columns
575        required_cols = {"genome_uuid", "meta_key", "desired_meta_value"}
576        if reader.fieldnames is None:
577            raise ValueError(f"CSV file {csv_file} has no header row")
578
579        if not required_cols.issubset(set(reader.fieldnames)):
580            raise ValueError(f"CSV must contain columns: {required_cols}")
581
582        for row_num, row in enumerate(reader, start=2):
583            # Check required fields
584            if not row.get("genome_uuid"):
585                logging.error(f"Row {row_num}: Missing required field genome_uuid")
586                continue
587
588            if not row.get("meta_key") or not row.get("desired_meta_value"):
589                logging.error(
590                    f"Row {row_num}: Missing required fields meta_key or desired_meta_value"
591                )
592                continue
593
594            patch = {
595                "genome_uuid": row["genome_uuid"].strip(),
596                "meta_key": row["meta_key"].strip(),
597                "desired_meta_value": row["desired_meta_value"].strip(),
598                "dataset_type": row.get("dataset_type", "genebuild").strip(),
599                "species_id": int(row.get("species_id", 1)),
600                "table_location": row.get(
601                    "table_location", "dataset_attribute"
602                ).strip(),
603                "core_suffix": core_suffix,
604                "row_num": row_num,
605            }
606
607            patches.append(patch)
608
609    return patches

612def check_thoas_requirements(patches: List[Dict], logger: logging.Logger) -> bool:
613    """
614    Check if any patches require THOAS (taxonomic heritage) updates.
615
616    Args:
617        patches: List of patch dictionaries
618        logger: Logger instance
619    Returns:
620        True if THOAS updates are required, False otherwise
621    """
622    thoas_loaded_fields = {
623        "assembly.accession",
624        "assembly.date",
625        "assembly.name",
626        "assembly.tol_id",
627        "organism.taxonomy_id",
628        "organism.species_taxonomy_id",
629        "organism.common_name",
630        "organism.scientific_name",
631        "organism.scientific_parlance_name",
632        "genome.genome_uuid",
633    }
634
635    for patch in patches:
636        if patch["meta_key"] in thoas_loaded_fields:
637            logger.info(
638                f"Patch for genome_uuid {patch['genome_uuid']} requires THOAS update"
639            )
640            logger.info(f"  Thoas-loaded field: {patch['meta_key']}")
641            return True
642    return False

645def resolve_genome_info(patch: Dict, logger: logging.Logger) -> Optional[Dict]:
646    """
647    Resolve production name from genome UUID.
648
649    Args:
650        patch: Patch dictionary from CSV
651        logger: Logger instance
652
653    Returns:
654        Dict with genome_uuid, production_name, core_db_name, or None if failed
655    """
656    row_num = patch["row_num"]
657    genome_uuid = patch["genome_uuid"]
658
659    logger.info(
660        f"Row {row_num}: Fetching production_name for genome_uuid: {genome_uuid}"
661    )
662    genome_info = get_genome_by_uuid(genome_uuid)
663
664    if not genome_info:
665        logger.error(
666            f"Row {row_num}: Could not fetch genome information for UUID: {genome_uuid}"
667        )
668        return None
669
670    production_name = genome_info["production_name"]
671    logger.info(f"Row {row_num}: Found production_name: {production_name}")
672
673    # Build core database name
674    core_db_name = f"{production_name}{patch['core_suffix']}"
675
676    return {
677        "genome_uuid": genome_uuid,
678        "production_name": production_name,
679        "core_db_name": core_db_name,
680    }

683def group_patches_by_genome(
684    patches: List[Dict], logger: logging.Logger, jira_ticket: str = ""
685) -> Dict[str, Dict]:
686    """
687    Group patches by genome UUID.
688
689    Args:
690        patches: List of patch dictionaries from CSV
691        logger: Logger instance
692        jira_ticket: Jira ticket reference
693
694    Returns:
695        Dict mapping genome_uuid to genome info and list of patches
696    """
697    grouped = {}
698
699    for patch in patches:
700        # Resolve genome information
701        genome_info = resolve_genome_info(patch, logger)
702
703        if not genome_info:
704            logger.warning(
705                f"Row {patch['row_num']}: Skipping due to resolution failure"
706            )
707            continue
708
709        genome_uuid = genome_info["genome_uuid"]
710
711        # Initialize genome entry if not exists
712        if genome_uuid not in grouped:
713            grouped[genome_uuid] = {
714                "genome_uuid": genome_uuid,
715                "production_name": genome_info["production_name"],
716                "core_db_name": genome_info["core_db_name"],
717                "dataset_type": patch["dataset_type"],
718                "species_id": patch["species_id"],
719                "jira_ticket": jira_ticket,
720                "patches": [],
721                "row_nums": [],
722            }
723
724        # Add patch to genome's list (include table_location)
725        grouped[genome_uuid]["patches"].append(
726            (patch["meta_key"], patch["desired_meta_value"], patch["table_location"])
727        )
728        grouped[genome_uuid]["row_nums"].append(patch["row_num"])
729
730    return grouped

733def generate_all_patches(  # pylint: disable=too-many-locals
734    grouped_patches: Dict[str, Dict],
735    output_dir: Path,
736    jira_ticket: str,
737    logger: logging.Logger,
738    team_filter: str = "Genebuild",
739) -> bool:
740    """
741    Generate consolidated patch files for all genomes.
742
743    Args:
744        grouped_patches: Dictionary of genome patches grouped by UUID
745        output_dir: Output directory for patch files
746        jira_ticket: Jira ticket reference
747        logger: Logger instance
748        team_filter: Optional team name to filter organism/assembly patches
749
750    Returns:
751        True if successful, False otherwise
752    """
753
754    # Metadata DB files
755    metadata_patch_file = output_dir / f"patch_metadata_{jira_ticket}.sql"
756    metadata_validate_file = output_dir / f"validate_metadata_{jira_ticket}.sql"
757
758    # Core DB files
759    core_patch_file = output_dir / f"patch_core_{jira_ticket}.sql"
760    core_validate_file = output_dir / f"validate_core_{jira_ticket}.sql"
761
762    try:
763        # Write metadata patches
764        with (
765            open(metadata_validate_file, "w") as val_f,
766            open(metadata_patch_file, "w") as patch_f,
767        ):
768            val_f.write(
769                f"-- Validation: Metadata DB | {jira_ticket} | {datetime.now().isoformat()}\n"
770            )
771            val_f.write("USE ensembl_genome_metadata;\n\n")
772
773            patch_f.write(
774                f"-- Patch: Metadata DB | {jira_ticket} | {datetime.now().isoformat()}\n"
775            )
776            patch_f.write(f"-- Validate first: {metadata_validate_file.name}\n")
777            patch_f.write("USE ensembl_genome_metadata;\n\n")
778
779            for genome_uuid, genome_data in grouped_patches.items():
780                production_name = genome_data["production_name"]
781                patches = genome_data["patches"]
782                logger.info(
783                    f"  Metadata: {production_name} ({genome_uuid}): {len(patches)} patches"
784                )
785
786                write_metadata_patch_for_genome(
787                    val_f,
788                    patch_f,
789                    genome_uuid,
790                    patches,
791                    genome_data["dataset_type"],
792                    logger,
793                    team_filter,
794                )
795
796        # Write core patches
797        with (
798            open(core_validate_file, "w") as val_f,
799            open(core_patch_file, "w") as patch_f,
800        ):
801            val_f.write(
802                f"-- Validation: Core DBs | {jira_ticket} | {datetime.now().isoformat()}\n\n"
803            )
804
805            patch_f.write(
806                f"-- Patch: Core DBs | {jira_ticket} | {datetime.now().isoformat()}\n"
807            )
808            patch_f.write(f"-- Validate first: {core_validate_file.name}\n\n")
809
810            for genome_uuid, genome_data in grouped_patches.items():
811                production_name = genome_data["production_name"]
812                core_db_name = genome_data["core_db_name"]
813                patches = genome_data["patches"]
814                logger.info(f"  Core: {core_db_name}: {len(patches)} patches")
815
816                write_core_patch_for_genome(
817                    val_f, patch_f, core_db_name, patches, genome_data["species_id"]
818                )
819
820        logger.info("Generated files:")
821        logger.info(f"  {metadata_validate_file}")
822        logger.info(f"  {metadata_patch_file}")
823        logger.info(f"  {core_validate_file}")
824        logger.info(f"  {core_patch_file}")
825
826        return True
827
828    except Exception as e:
829        logger.error(f"Failed to generate patches: {e}")
830        return False

833def main():
834    """Main entry point for the beta patcher script."""
835    parser = argparse.ArgumentParser(
836        description="Generate standardized SQL patches for Ensembl\
837            beta metadata issues from CSV input",
838        formatter_class=argparse.RawDescriptionHelpFormatter,
839        epilog="""
840Examples:
841# Process patches from CSV file
842python beta_patcher.py patches.csv --jira-ticket EBD-1111 --output-dir ./patches/
843
844# Process CSV with custom core suffix
845python beta_patcher.py patches.csv --jira-ticket EBD-1111 --core-suffix _core_115_1
846
847# Validation files are automatically generated
848# Run validate_*.sql files before applying patch_*.sql files
849
850CSV Format:
851  Required columns: genome_uuid, meta_key, desired_meta_value
852  Optional columns: dataset_type, species_id, table_location
853
854  table_location valid values: dataset_attribute (default), genome, organism, assembly
855
856  Example:
857    genome_uuid,meta_key,desired_meta_value,dataset_type,species_id,table_location
858    a7335667-93e7-11ec-a39d-005056b38ce3,assembly.name,GRCh38.p14,genebuild,1,dataset_attribute
859    a7335667-93e7-11ec-a39d-005056b38ce3,genebuild_version,2024-01,genebuild,1,genome
860    a7335667-93e7-11ec-a39d-005056b38ce3,strain,reference,genebuild,1,organism
861    a7335667-93e7-11ec-a39d-005056b38ce3,level,chromosome,genebuild,1,assembly
862
863See patches_template.csv for a complete example.
864        """,
865    )
866
867    # Positional argument
868    parser.add_argument(
869        "csv_file",
870        type=Path,
871        help="CSV file with patches (see patches_template.csv for format)",
872    )
873
874    # Jira ticket (required)
875    parser.add_argument(
876        "--jira-ticket",
877        type=str,
878        required=True,
879        help="Jira ticket reference (e.g., EBD-1111)",
880    )
881
882    # Core database suffix
883    parser.add_argument(
884        "--core-suffix",
885        type=str,
886        default="_core_114_1",
887        help="Suffix to append to production_name for core DB name (default: _core_114_1)",
888    )
889
890    # Database connection (uses environment variables METADATA_URI and TAXONOMY_URI)
891    parser.add_argument(
892        "--metadata-uri",
893        type=str,
894        help="Metadata database URI (overrides METADATA_URI env var)",
895    )
896
897    parser.add_argument(
898        "--taxonomy-uri",
899        type=str,
900        help="Taxonomy database URI (overrides TAXONOMY_URI env var)",
901    )
902
903    # Output options
904    parser.add_argument(
905        "-o",
906        "--output-dir",
907        type=Path,
908        default=Path.cwd(),
909        help="Output directory for patch files (default: current directory)",
910    )
911
912    # Team filter option
913    parser.add_argument(
914        "--team-filter",
915        type=str,
916        help='Only apply patches if affected genomes belong to this team (e.g., "genebuild")',
917    )
918
919    args = parser.parse_args()
920
921    # Validate CSV file exists
922    if not args.csv_file.exists():
923        parser.error(f"CSV file not found: {args.csv_file}")
924
925    # Override environment variables if provided
926    if args.metadata_uri:
927        os.environ["METADATA_URI"] = args.metadata_uri
928    if args.taxonomy_uri:
929        os.environ["TAXONOMY_URI"] = args.taxonomy_uri
930
931    # Create output directory
932    args.output_dir.mkdir(parents=True, exist_ok=True)
933
934    # Set up logging
935    log_file = (
936        args.output_dir / f"patch_csv_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log"
937    )
938    logging.basicConfig(
939        level=logging.INFO,
940        format="%(asctime)s - %(levelname)s - %(message)s",
941        handlers=[logging.FileHandler(log_file), logging.StreamHandler(sys.stdout)],
942    )
943    logger = logging.getLogger(__name__)
944
945    try:
946        patches = read_csv_patches(args.csv_file, args.core_suffix)
947        logger.info(f"Processing {len(patches)} patches from {args.csv_file}")
948    except Exception as e:
949        logger.error(f"Failed to read CSV: {e}")
950        return 1
951
952    if check_thoas_requirements(patches, logger):
953        logger.error(
954            "One or more patches require THOAS updates. Please handle these separately."
955        )
956        return 1
957
958    grouped_patches = group_patches_by_genome(patches, logger, args.jira_ticket)
959    logger.info(f"Grouped into {len(grouped_patches)} genome(s)")
960
961    if not grouped_patches:
962        logger.error("No valid patches to process after grouping")
963        return 1
964
965    # Generate consolidated patch files
966    team_filter = getattr(args, "team_filter", None)
967    if generate_all_patches(
968        grouped_patches, args.output_dir, args.jira_ticket, logger, team_filter
969    ):
970        logger.info(f"Success! Log: {log_file}")
971        return 0
972
973    logger.error(f"Failed. Log: {log_file}")
974    return 1

978    sys.exit(main())
