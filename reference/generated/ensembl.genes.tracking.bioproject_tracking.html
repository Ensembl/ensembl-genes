 55def mysql_fetch_data(
 56    query: str,
 57    params: Tuple = (),
 58    server_group: str = "meta",
 59    server_name: str = "beta",
 60    db_name: Optional[str] = None,
 61) -> Sequence[Tuple[Any, ...]]:
 62    """
 63    Executes a SQL query with optional parameters to fetch results from a specified MySQL server.
 64
 65    Args:
 66        query (str): The SQL query to execute.
 67        params (Tuple, optional): Parameters to pass to the SQL query. Defaults to ().
 68        server_group (str, optional): The server group as defined in the config. Defaults to "meta".
 69        server_name (str, optional): The server name within the group as defined in the config. Defaults to "beta".
 70        db_name (Optional[str], optional): The database name to connect to. If None, uses the default from config. Defaults to None.
 71
 72    Returns:
 73        Sequence[Tuple[Any, ...]]: The fetched rows from the query result.
 74
 75    Raises:
 76        KeyError: If the server group or name is not found in the config.
 77        pymysql.Error: If there is an error connecting to the database or executing the query.
 78    """
 79    try:
 80        server_config = config["server_details"][server_group][server_name]
 81        connection = pymysql.connect(
 82            host=server_config["db_host"],
 83            user=server_config["db_user"],
 84            port=server_config["db_port"],
 85            database=db_name or server_config.get("db_name", ""),
 86        )
 87        with connection.cursor() as cursor:
 88            cursor.execute(query, params)
 89            result = cursor.fetchall()
 90        connection.close()
 91        return result
 92
 93    except KeyError as key_err:
 94        logging.error(f"Invalid server group or name in config: {key_err}")
 95    except pymysql.Error as sql_err:
 96        logging.error(f"MySQL Error: {sql_err}")
 97
 98    return []

104def get_assembly_accessions(  # pylint:disable=too-many-branches, too-many-statements, too-many-locals
105    query_id: str, query_type: str, only_haploid: bool = False
106) -> Dict[str, Dict[str, int]]:
107    """
108    Fetches assembly accessions from NCBI Datasets API based on BioProject or Taxon ID.
109
110    Args:
111        query_id (str): The BioProject or Taxon ID to query.
112        query_type (str): The type of query, either "bioproject" or "taxon".
113        only_haploid (bool): If True, only include haploid assemblies. Defaults to False.
114
115    Returns:
116        Dict[str, Dict[str, int]]: A dictionary mapping assembly accessions to their taxon IDs.
117    """
118
119    def _parse_reports(lines: List[str]) -> Dict[str, Dict[str, int]]:
120        accs: Dict[str, Dict[str, int]] = {}
121        for line in lines:
122            line = line.strip()
123            if not line:
124                continue
125            try:
126                obj = json.loads(line)
127            except json.JSONDecodeError:
128                continue
129            report = obj.get("report", obj)
130            assembly_info = report.get("assembly_info") or {}
131            if only_haploid and assembly_info.get("assembly_type") != "haploid":
132                continue
133            accession = report.get("accession")
134            taxon_id = (report.get("organism") or {}).get("tax_id")
135            if not isinstance(taxon_id, int):
136                continue
137            if accession:
138                accs[accession] = {"taxon_id": taxon_id}
139        return accs
140
141    # --- 1) Try CLI (gets ALL versions) ---
142    datasets_cli = shutil.which("datasets")
143    if datasets_cli:
144        try:
145            if query_type == "bioproject":
146                cmd = [
147                    datasets_cli,
148                    "summary",
149                    "genome",
150                    "accession",
151                    str(query_id),
152                    "--assembly-version",
153                    "all",
154                    "--as-json-lines",
155                ]
156            elif query_type == "taxon":
157                cmd = [
158                    datasets_cli,
159                    "summary",
160                    "genome",
161                    "taxon",
162                    str(query_id),
163                    "--assembly-version",
164                    "all",
165                    "--as-json-lines",
166                ]
167            else:
168                logging.error(f"Invalid query_type '{query_type}'")
169                return {}
170
171            result = subprocess.run(cmd, capture_output=True, text=True, check=True)
172            parsed = _parse_reports(result.stdout.splitlines())
173            if parsed:  # pylint:disable=no-else-return
174                logging.info(
175                    f"[Datasets CLI] Retrieved {len(parsed)} assemblies (all versions)."
176                )
177                return parsed
178            else:
179                logging.warning(
180                    "[Datasets CLI] Parsed zero rows from stdout; falling back to API."
181                )
182        except subprocess.CalledProcessError as e:
183            logging.warning(
184                f"[Datasets CLI] Exit {e.returncode}. STDERR:\n{e.stderr}\nFalling back to API."
185            )
186        except Exception as e:  # pylint:disable=broad-exception-caught
187            logging.warning(
188                f"[Datasets CLI] Unexpected error: {e}. Falling back to API."
189            )
190
191    # --- 2) Fallback: API (latest only) ---
192    try:
193        base_url = config["urls"]["datasets"].get(query_type)
194    except Exception:  # pylint:disable=broad-exception-caught
195        base_url = None
196
197    if not base_url:
198        logging.error(
199            "[Datasets API] Base URL not found in config; returning empty set."
200        )
201        return {}
202
203    assembly_accessions: Dict[str, Dict[str, int]] = {}
204    page_size = 5000
205    next_page_token = None
206
207    while True:
208        url = f"{base_url}/{query_id}/dataset_report?page_size={page_size}"
209        if next_page_token:
210            url += f"&page_token={next_page_token}"
211        try:
212            response = requests.get(url)  # pylint:disable=missing-timeout
213            response.raise_for_status()
214            data = response.json()
215            for assembly in data.get("reports", []):
216                assembly_info = assembly.get("assembly_info", {}) or {}
217                if only_haploid and assembly_info.get("assembly_type") != "haploid":
218                    continue
219                accession = assembly.get("accession")
220                taxon_id = (assembly.get("organism") or {}).get("tax_id")
221                if not isinstance(taxon_id, int):
222                    continue
223                if accession:
224                    assembly_accessions[accession] = {"taxon_id": taxon_id}
225            next_page_token = data.get("next_page_token")
226            if not next_page_token:
227                break
228        except requests.RequestException as e:
229            logging.error(f"[Datasets API] Error: {e}")
230            break
231
232    logging.info(
233        f"[Datasets API] Retrieved {len(assembly_accessions)} assemblies (latest only)."
234    )
235    return assembly_accessions

241_CORE_VER_RE = re.compile(r"_core_(\d+)_", re.IGNORECASE)
242
243
244def _score_dbname(dbname: str) -> Tuple[int, int, str]:
245    """
246    Higher tuple is better.
247    1) Higher core version
248    2) Prefer non-'cm'
249    3) Lexicographic tiebreak
250
251    Args:
252        dbname (str): The database name to score.
253
254    Returns:
255        Tuple[int, int, str]: A tuple representing the score.
256    """
257    m = _CORE_VER_RE.search(dbname or "")
258    core_ver = int(m.group(1)) if m else -1
259    non_cm = 1 if "_cm_" not in (dbname or "").lower() else 0
260    return (core_ver, non_cm, dbname or "")

263def get_ensembl_live(
264    accessions_taxon: Dict[str, Dict[str, int]],
265) -> Tuple[Dict[str, Dict[str, Any]], List[str]]:
266    """
267    Fetches Ensembl live database annotations for a list of assembly accessions.
268
269    Args:
270        accessions_taxon (Dict[str, Dict[str, int]]): A dictionary mapping assembly accessions
271            to basic taxon information, such as {"GCA_000001405.39": {"taxon_id": 9606}}.
272
273    Returns:
274        Tuple[Dict[str, Dict[str, Any]], List[str]]: A tuple containing:
275        1) A dictionary where each key is an assembly accession and its value is another dictionary with:
276           - taxon_id (from input)
277           - matches: List[{"guuid","dbname"}]
278           - guuid, dbname: best match (latest core; prefer non-cm)
279        2) A list of accessions that were not found in the Ensembl live database.
280    """
281    accessions = list(accessions_taxon.keys())
282    if not accessions:
283        return {}, []
284    placeholders = ", ".join(["%s"] * len(accessions))
285    query = f"""
286    SELECT DISTINCT
287        assembly.accession,
288        genome.genome_uuid,
289        dataset_source.name AS database_name
290    FROM genome
291    JOIN assembly ON genome.assembly_id = assembly.assembly_id
292    JOIN genome_dataset ON genome.genome_id = genome_dataset.genome_id
293    JOIN dataset ON genome_dataset.dataset_id = dataset.dataset_id
294    JOIN dataset_source ON dataset.dataset_source_id = dataset_source.dataset_source_id
295    JOIN genome_release ON genome.genome_id = genome_release.genome_id
296    JOIN ensembl_release ON genome_release.release_id = ensembl_release.release_id
297    WHERE dataset.name = 'genebuild'
298      AND assembly.accession IN ({placeholders})
299    """
300
301    rows = mysql_fetch_data(query, tuple(accessions))
302
303    # Seed with taxon info so missing ones can still be tracked
304    live_annotations: Dict[str, Dict[str, Any]] = {}
305
306    for accession, guuid, dbname in rows:
307        entry = live_annotations.setdefault(
308            accession, dict(accessions_taxon.get(accession, {}))
309        )
310        entry.setdefault("matches", [])
311        # Avoid duplicates across joins if any
312        if not any(
313            m["guuid"] == guuid and m["dbname"] == dbname for m in entry["matches"]
314        ):
315            entry["matches"].append({"guuid": guuid, "dbname": dbname})
316
317    # Choose best match for backward compatibility fields
318    for accession, entry in live_annotations.items():
319        matches = entry.get("matches", [])
320        if matches:
321            best = max(matches, key=lambda m: _score_dbname(m["dbname"]))
322            entry["guuid"] = best["guuid"]
323            entry["dbname"] = best["dbname"]
324
325    # Compute missing (accessions with no rows at all)
326    matched_accessions = set(live_annotations.keys())
327    missing_annotations = [acc for acc in accessions if acc not in matched_accessions]
328
329    return live_annotations, missing_annotations

335def get_taxonomy_info(
336    live_annotations: Dict[str, Dict[str, Any]],
337    accessions_taxon: Dict[str, Dict[str, int]],
338    rank: str,
339) -> Dict[str, Dict[str, Any]]:
340    """
341    Fetches taxonomy information for a given rank from the NCBI Datasets API.
342    For each accession in `live_annotations`, this function retrieves its `taxon_id`\
343    and queries the NCBI taxonomy endpoint to get the classification at the specified rank.
344    The retrieved rank name is then added to the corresponding annotation data.
345    
346    Args:
347        live_annotations (Dict[str, Dict[str, str]]): A dictionary where each key is \
348            an assembly accession and its value contains annotation details (including \
349            at least "taxon_id").
350        accessions_taxon (Dict[str, Dict[str, int]]): A dictionary mapping accessions \
351            to basic taxon information, such as {"GCA_000001405.39": {"taxon_id": 9606}}.
352        rank (str): The taxonomic rank to retrieve (e.g., "order", "class", "phylum").
353        
354    Returns:
355        Dict[str, Dict[str, str]]: The updated `live_annotations` dictionary with an
356        additional key for the requested rank. For example:
357            {
358                "GCA_000001405.39": {
359                    "taxon_id": 9606,
360                    "guuid": "some-uuid",
361                    "dbname": "ensembl_core",
362                    "order": "Primates"
363                },
364                ...
365            }
366        If the API query fails or the rank is not found, the value for that rank is not added
367        (or set to None/unknown).
368    """
369    for accession in live_annotations:
370        taxon_id = accessions_taxon[accession]["taxon_id"]
371        url = f"https://api.ncbi.nlm.nih.gov/datasets/v2alpha/taxonomy/taxon/{taxon_id}/dataset_report"
372
373        try:
374            response = requests.get(url)  # pylint:disable=missing-timeout
375            response.raise_for_status()
376            data = response.json()
377
378            taxonomies = data.get("reports", [])
379            for taxonomy in taxonomies:
380                rank_name = (
381                    taxonomy.get("taxonomy", {})
382                    .get("classification", {})
383                    .get(rank, {})
384                    .get("name")
385                )
386                taxonomy_info = {rank: rank_name}
387                live_annotations[accession].update(taxonomy_info)
388
389        except requests.HTTPError as http_err:
390            logging.error(f"HTTP error occurred: {http_err}")
391        except Exception as err:  # pylint:disable=broad-exception-caught
392            logging.error(f"An error occurred: {err}")
393
394    return live_annotations

400def add_ftp(  # pylint:disable=too-many-locals, too-many-branches
401    annotations: Dict[str, Dict[str, Any]], release_type: str = "live"
402) -> Dict[str, Dict[str, Any]]:
403    """
404    Adds FTP links to the annotations dictionary based on the release type.
405    
406    Args:
407        annotations (Dict[str, Dict[str, Any]]): A dictionary where each key is \
408            an assembly accession and its value contains annotation details, \
409            including matches with database names or genome UUIDs.
410        release_type (str): The type of release to determine FTP link construction.\
411            Can be either "live" or "pre".
412            
413    Returns:
414        Dict[str, Dict[str, Any]]: The updated annotations dictionary with FTP links \
415        added to each match and the top-level annotation where applicable.
416    """
417    for accession, annotation in annotations.items():
418        matches = annotation.get("matches") or []
419
420        if release_type == "pre":
421            dbnames = set(m.get("dbname") for m in matches if m.get("dbname"))
422            if annotation.get("dbname"):
423                dbnames.add(annotation["dbname"])
424
425            for dbname in dbnames:
426                query = """
427                SELECT meta_value
428                FROM meta
429                WHERE meta_key = 'organism.scientific_name'
430                """
431                result = mysql_fetch_data(
432                    query,
433                    (),
434                    server_group="pre-release",
435                    server_name="gb1",
436                    db_name=dbname,
437                )
438
439                if not result:
440                    continue  # skip if no scientific name found
441
442                scientific_name = result[0][0].replace(" ", "_").replace(".", "")
443                ftp_link = f"https://ftp.ebi.ac.uk/pub/databases/ensembl/pre-release/{scientific_name}/{accession}"
444
445                # attach to matching entries
446                for m in matches:
447                    if m.get("dbname") == dbname:
448                        m["ftp"] = ftp_link
449                if annotation.get("dbname") == dbname:
450                    annotation["ftp"] = ftp_link  # top-level
451
452        elif release_type == "live":
453            # Build map guuid->match for easy updates
454            by_guuid = {m.get("guuid"): m for m in matches if m.get("guuid")}
455            guuids = set(by_guuid.keys())
456            if annotation.get("guuid"):
457                guuids.add(annotation["guuid"])
458
459            for guuid in guuids:
460                if not guuid or guuid == "unknown":
461                    continue
462                query = """
463                SELECT genome.genebuild_date, organism.scientific_name, dataset_attribute.value
464                FROM genome
465                JOIN organism USING(organism_id)
466                JOIN genome_dataset USING(genome_id) 
467                JOIN dataset_attribute USING(dataset_id)
468                WHERE genome.genome_uuid = %s
469                AND attribute_id=169
470                """
471                data_fetch = mysql_fetch_data(query, (guuid,))
472                if not data_fetch:
473                    continue
474
475                date, scientific_name, source = data_fetch[0]
476                date = str(date).replace("-", "_")
477                scientific_name = scientific_name.replace(" ", "_").replace(".", "")
478                source = source.lower()
479                ftp_link = f"https://ftp.ebi.ac.uk/pub/ensemblorganisms/{scientific_name}/{accession}/{source}/geneset/{date}/"
480
481                if guuid in by_guuid:
482                    by_guuid[guuid]["ftp"] = ftp_link
483                    by_guuid[guuid]["date"] = date
484                if annotation.get("guuid") == guuid:
485                    annotation["ftp"] = ftp_link
486                    annotation["date"] = date
487
488    return annotations

491def get_pre_release(missing_annotations: List[str]) -> Dict[str, Dict[str, str]]:
492    """
493    Checks for the existence of pre-release Ensembl database schemas for a \
494        list of missing accessions.
495    This function takes a list of GenBank/RefSeq assembly accessions that were not found in the \
496    Ensembl live database lookup (e.g., from `get_ensembl_live`). It reformats each accession \
497    to match the expected pre-release Ensembl database schema naming convention (e.g.,\
498    'GCA_000001405.39' → 'gca000001405v39') and queries the MySQL server's \
499    `information_schema.schemata` table to check if a database with that name exists.
500    If such a schema is found, the function records the corresponding accession and \
501    database name in the returned dictionary.
502    
503    Args:
504        missing_annotations (List[str]): A list of assembly accessions (e.g., 'GCA_000001405.39') \
505            for which no live annotation database was found.
506            
507    Returns:
508        Dict[str, Dict[str, str]]: A dictionary mapping each accession with a found pre-release 
509        database to a nested dictionary containing:
510            - "dbname": The name of the pre-release schema (e.g., 'gca000001405v39').
511        Example:
512            {
513                "GCA_000001405.39": {
514                    "dbname": "gca000001405v39"
515                },
516                ...
517            }
518    Notes:
519        - Only accessions matching the pattern 'GCA/GCF_XXXXXXXXX.XX' are processed.
520        - If no matching schema is found for an accession, it is omitted from the result.
521        - The function uses `mysql_fetch_data` to query the MySQL server's schema metadata.
522    """
523    pre_release_annotations: Dict[str, Dict[str, str]] = {}
524
525    for accession in missing_annotations:
526        match = re.match(r"(GCA|GCF)_(\d+)\.(\d+)", accession)
527        if not match:
528            continue  # skip malformed accession
529
530        prefix, number, version = match.groups()
531        lookup_string = f"%{prefix.lower()}{number}v{version}%"
532
533        query = """
534        SELECT SCHEMA_NAME
535        FROM information_schema.schemata
536        WHERE SCHEMA_NAME like %s
537        """
538        result = mysql_fetch_data(
539            query, (lookup_string,), server_group="pre-release", server_name="gb1"
540        )
541
542        if result:
543            pre_release_annotations[accession] = {
544                "guuid": "unknown",  # typo fixed from guiid -> guuid
545                "dbname": result[0][0],
546            }
547    return pre_release_annotations

553def write_report(
554    live_annotations: Dict[str, Dict[str, Any]],
555    report_file: str,

558    include_ftp: bool = False,
559):
560    """

563    
564    Args:
565        live_annotations (Dict[str, Dict[str, Any]]): A dictionary where each key is \
566            an assembly accession and its value contains annotation details, \
567            including matches with database names or genome UUIDs.
568        report_file (str): The path to the output report file.
569        rank (str): The taxonomic rank to include if `include_class` is True.
570        include_class (bool): If True, includes the taxonomic classification at the specified rank.
571        include_ftp (bool): If True, includes FTP links in the report.
572        
573    Returns:
574        None
575    """
576    report_path = Path(report_file)
577    lines_written = 0
578
579    with open(report_path, "w", encoding="utf-8") as file:
580        for accession, details in live_annotations.items():
581            matches = details.get("matches")
582
583            # If we have multiple, write one row per match
584            rows = (
585                matches
586                if matches
587                else [
588                    {
589                        "guuid": details.get("guuid"),
590                        "dbname": details.get("dbname"),
591                        "ftp": details.get("ftp"),
592                        "date": details.get("date"),
593                    }
594                ]
595            )
596
597            for m in rows:
598                row = [
599                    str(accession),
600                    str(m.get("guuid") or "unknown"),
601                    str(m.get("dbname") or "unknown"),
602                ]
603                if include_class:
604                    row.append(str(details.get(rank) or "unknown"))
605                if include_ftp:
606                    row.append(str(m.get("date") or details.get("date") or ""))
607                    row.append(str(m.get("ftp") or details.get("ftp") or "N/A"))
608
609                file.write("\t".join(row) + "\n")
610                lines_written += 1
611
612    logging.info(
613        f"Report written to {report_path.resolve()} with {lines_written} lines."
614    )
615

617def main():
618    """
619    Main entry point of the script.
620    Parses command-line arguments to determine if a BioProject ID or Taxon ID is provided,
621    whether to filter only haploid assemblies, the desired output file path, the taxonomic
622    rank to retrieve, and whether to include FTP links. It then:
623      1. Fetches assembly accessions from the NCBI API based on the provided ID.
624      2. Fetches corresponding Ensembl live database information from the local MySQL database.
625      3. Retrieves taxonomy information for the specified rank from the NCBI API.
626      4. Optionally adds FTP links if requested.
627      5. Writes all collected data into a tab-separated report file.
628    Usage Example:
629        python script_name.py --bioproject_id PRJNA12345 --haploid --rank order --ftp
630    Returns:
631        None
632    """
633    parser = argparse.ArgumentParser(
634        description="Fetch assembly data and report annotations."
635    )
636    parser.add_argument("--bioproject_id", type=str, help="NCBI BioProject ID")
637    parser.add_argument("--taxon_id", type=str, help="Taxonomy ID")
638    parser.add_argument(
639        "--haploid", action="store_true", help="Fetch only haploid assemblies"
640    )
641    parser.add_argument("--report_file", type=str, default="./report_file.tsv")
642    parser.add_argument(
643        "--classification",
644        action="store_true",
645        help="Provide breakdown of taxonomic classification",
646    )
647    parser.add_argument("--rank", type=str, default="order")
648    parser.add_argument(
649        "--ftp", action="store_true", help="Include FTP links in the report"
650    )
651    parser.add_argument(
652        "--pre_release",
653        action="store_true",
654        help="Include list of pre-release databases in the report",
655    )
656
657    args = parser.parse_args()
658
659    # Determine which ID is provided and which query type is relevant
660    if not args.bioproject_id and not args.taxon_id:
661        parser.error("You must provide either --bioproject_id or --taxon_id")
662
663    accessions_taxon = get_assembly_accessions(
664        args.bioproject_id or args.taxon_id,
665        "bioproject" if args.bioproject_id else "taxon",
666        args.haploid,
667    )
668
669    # Fetch annotations (live)
670    live_annotations, missing_annotations = get_ensembl_live(accessions_taxon)
671
672    # Taxonomy classification (top-level only; rows inherit from details)
673    if args.classification:
674        get_taxonomy_info(live_annotations, accessions_taxon, args.rank)
675
676    # Optionally add FTP links
677    if args.ftp:
678        live_annotations = add_ftp(live_annotations, "live")
679
680    # Unique species count from live annotations
681    unique_taxon_ids = {
682        details["taxon_id"]
683        for details in live_annotations.values()
684        if "taxon_id" in details
685    }
686
687    # Optionally add Pre-release data
688    if args.pre_release and missing_annotations:
689        pre_release_annotations = get_pre_release(missing_annotations)
690
691        for (
692            accession
693        ) in pre_release_annotations:  # pylint:disable=consider-using-dict-items
694            if accession in accessions_taxon:
695                pre_release_annotations[accession]["taxon_id"] = accessions_taxon[
696                    accession
697                ]["taxon_id"]
698
699        if args.classification and pre_release_annotations:
700            get_taxonomy_info(pre_release_annotations, accessions_taxon, args.rank)
701
702        if args.ftp and pre_release_annotations:
703            pre_release_annotations = add_ftp(pre_release_annotations, "pre")
704
705        all_annotations: Dict[str, Dict[str, Any]] = {  # type: ignore
706            **live_annotations,
707            **pre_release_annotations,
708        }
709    else:
710        all_annotations = live_annotations
711
712    # Console summary
713    if args.bioproject_id:
714        print(
715            f"Found {len(accessions_taxon)} assemblies under BioProject ID {args.bioproject_id}"
716        )
717    elif args.taxon_id:
718        print(f"Found {len(accessions_taxon)} assemblies for taxon ID {args.taxon_id}")
719
720    print(
721        f"Found {len(live_annotations)} annotations in beta.ensembl.org \
722            for {len(unique_taxon_ids)} unique species"
723    )
724
725    if args.classification:
726        rank_values = [
727            details.get(args.rank, "unknown") for details in live_annotations.values()
728        ]
729        rank_counts = Counter(rank_values)
730        print("\nBreakdown:")
731        print(rank_counts)
732
733    # Write final report
734    write_report(
735        all_annotations,
736        args.report_file,
737        args.rank,
738        include_class=args.classification,
739        include_ftp=args.ftp,
740    )
741
742
743if __name__ == "__main__":
744    main()
